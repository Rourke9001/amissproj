**Goal:** Apartment tiers (Low-Cost / Le Security) with wiki-exact rent, a paid-through lease with advances, rent extensions, apartment switching, garnishment fidelity, and tier-aware weekly start position — per the approved spec `docs/superpowers/specs/2026-07-19-kan50-housing-design.md`.

package amiss.domain.model;

import amiss.domain.board.Location;

/**
 * The two rentable apartments (KAN-50), wiki-exact. {@code baseRent} is the tuning
 * constant every listing prices from via {@code EconomyService.price}; the rent a save
 * actually pays is its locked-in {@code SaveState.rentMonthly}. {@code robberyProof}
 * is the entire KAN-24 robbery hook. {@code homeStop} is where each week starts.
 */
public enum ApartmentTier {

    LOW_COST(325, false, Location.LOW_COST_HOUSING),
    SECURITY(475, true, Location.LE_SECURITY_APARTMENTS);

    private final int baseRent;
    private final boolean robberyProof;
    private final Location homeStop;

    ApartmentTier(int baseRent, boolean robberyProof, Location homeStop) {
        this.baseRent = baseRent;
        this.robberyProof = robberyProof;
        this.homeStop = homeStop;
    }

    public int baseRent() {
        return baseRent;
    }

    public boolean robberyProof() {
        return robberyProof;
    }

    public Location homeStop() {
        return homeStop;
    }

    /** The one tier you can switch to (wiki: only direct type-to-type switches exist). */
    public ApartmentTier other() {
        return this == LOW_COST ? SECURITY : LOW_COST;
    }
}

```java
package amiss.application.service.save;

import amiss.application.config.ActionCosts;
import amiss.application.port.JobCatalog;
import amiss.application.port.SaveRepository;
import amiss.domain.board.Location;
import amiss.domain.model.ApartmentTier;
import amiss.domain.model.SaveState;
import java.util.function.IntSupplier;

/**
 * The Rent Office rules (KAN-50), wiki-exact: a paid-through monthly lease (due every
 * 4th round), pre-pay advances, one-week extensions with decaying approval odds,
 * apartment switching at the floating listing price, and debt payoff. The office is
 * open by calendar — month-end weeks — or while the save holds an extension; a Rent
 * Office employee can enter any week but gets no services (wiki NOTE). The signed
 * rent is locked; only listings float with the economy.
 */
public class RentService {

    private static final int MONTH_ROUNDS = 4;
    /** Approval % by extensions already granted (index clamped to 3 — the wiki floor). */
    private static final int[] EXTENSION_APPROVAL_PCT = {100, 75, 50, 25};
    private static final int EXTENSION_DENIED_HAPPINESS = -1;
    private static final int EXTENSION_GRACE_ROUNDS = 1;

    private final SaveRepository saves;
    private final JobCatalog jobs;
    private final ActionCosts costs;
    private final EconomyService economy;
    private final IntSupplier roll1to100;

    public RentService(SaveRepository saves, JobCatalog jobs, ActionCosts costs,
            EconomyService economy, IntSupplier roll1to100) {
        this.saves = saves;
        this.jobs = jobs;
        this.costs = costs;
        this.economy = economy;
        this.roll1to100 = roll1to100;
    }

    /** The smallest month-end (multiple of 4) strictly after {@code round}. */
    public static int nextMonthEndAfter(int round) {
        return (round / MONTH_ROUNDS + 1) * MONTH_ROUNDS;
    }

    /** Services are available on month-end weeks or while on extension (wiki). */
    public boolean servicesOpen(SaveState save) {
        return save.round() % MONTH_ROUNDS == 0 || save.onExtension();
    }

    /** The building is enterable when services run, or any week for its employees. */
    public boolean officeOpen(SaveState save) {
        return servicesOpen(save) || worksAtRentOffice(save);
    }

    private boolean worksAtRentOffice(SaveState save) {
        return save.employed() && jobs.byId(save.jobId())
                .map(job -> Location.RENT_OFFICE.displayName().equals(job.location()))
                .orElse(false);
    }

    /** The floating market price to sign {@code tier} today (crash can halve it). */
    public int listing(ApartmentTier tier, SaveState save) {
        return economy.price(tier.baseRent(), save);
    }

    public RentView view(SaveState save) {
        boolean services = servicesOpen(save);
        boolean canAsk = services && save.rentDue() && !save.everRentDebt()
                && !alreadyAskedThisRound(save);
        return new RentView(officeOpen(save), services, save.rentDue(),
                save.apartmentTier(), save.rentMonthly(), save.leasePaidThrough(),
                save.extensionUntil(), canAsk, save.debt(),
                listing(ApartmentTier.LOW_COST, save), listing(ApartmentTier.SECURITY, save));
    }

    private boolean alreadyAskedThisRound(SaveState save) {
        return save.extensionAskedRound() != null && save.extensionAskedRound() == save.round();
    }

    /**
     * Pay 1 Month of Rent: clears the whole debt if any (wiki: "erase your entire Rent
     * Debt"), else pays the due month, else buys an advance month.
     */
    public RentPayment payRent(SaveState save) {
        if (save.debt() > 0) {
            return payDebt(save);
        }
        RentPayment rejection = rejectPayment(save, save.rentMonthly());
        if (rejection != null) {
            return rejection;
        }
        charge(save, save.rentMonthly());
        RentPayment.Status status;
        if (save.rentDue()) {
            save.setLeasePaidThrough(nextMonthEndAfter(save.round()));
            save.setExtensionUntil(null);
            status = RentPayment.Status.OK_MONTH;
        } else {
            save.setLeasePaidThrough(save.leasePaidThrough() + MONTH_ROUNDS);
            status = RentPayment.Status.OK_ADVANCE;
        }
        saves.update(save);
        return new RentPayment(status, save.rentMonthly(), save.leasePaidThrough(),
                save.timeMinutes(), save.cash());
    }

    /** Pay Garnishment: the whole debt, in cash, in one go. */
    public RentPayment payDebt(SaveState save) {
        int debt = save.debt();
        if (debt == 0) {
            return new RentPayment(RentPayment.Status.NOT_IN_DEBT, 0,
                    save.leasePaidThrough(), save.timeMinutes(), save.cash());
        }
        RentPayment rejection = rejectPayment(save, debt);
        if (rejection != null) {
            return rejection;
        }
        charge(save, debt);
        save.setDebt(0);
        saves.update(save);
        return new RentPayment(RentPayment.Status.OK_DEBT_CLEARED, debt,
                save.leasePaidThrough(), save.timeMinutes(), save.cash());
    }

    /** The shared guards; null when the payment may proceed. Nothing is charged here. */
    private RentPayment rejectPayment(SaveState save, int amount) {
        RentPayment.Status status = null;
        if (!servicesOpen(save)) {
            status = RentPayment.Status.OFFICE_CLOSED;
        } else if (save.weekOver()) {
            status = RentPayment.Status.WEEK_OVER;
        } else if (save.cash() < amount) {
            status = RentPayment.Status.INSUFFICIENT_CASH;
        } else if (save.timeMinutes() < costs.payRentMinutes()) {
            status = RentPayment.Status.INSUFFICIENT_TIME;
        }
        return status == null ? null
                : new RentPayment(status, 0, save.leasePaidThrough(), save.timeMinutes(), save.cash());
    }

    private void charge(SaveState save, int amount) {
        save.setTimeMinutes(save.timeMinutes() - costs.payRentMinutes());
        save.setCash(save.cash() - amount);
    }

    /**
     * Ask for a one-week extension. Once per round (persisted guard); auto-denied
     * forever once the save has been in Rent Debt (wiki). Approval odds decay
     * 100/75/50/25 with each granted extension, floor 25.
     */
    public RentExtension requestExtension(SaveState save) {
        if (!servicesOpen(save)) {
            return new RentExtension(RentExtension.Status.OFFICE_CLOSED, save.extensionUntil());
        }
        if (save.weekOver()) {
            return new RentExtension(RentExtension.Status.WEEK_OVER, save.extensionUntil());
        }
        if (!save.rentDue()) {
            return new RentExtension(RentExtension.Status.NOT_DUE, save.extensionUntil());
        }
        if (alreadyAskedThisRound(save)) {
            return new RentExtension(RentExtension.Status.ALREADY_ASKED, save.extensionUntil());
        }
        save.setExtensionAskedRound(save.round());
        boolean approved = false;
        if (!save.everRentDebt()) {
            int pct = EXTENSION_APPROVAL_PCT[Math.min(save.extensionsGranted(),
                    EXTENSION_APPROVAL_PCT.length - 1)];
            approved = pct == 100 || roll1to100.getAsInt() <= pct;
        }
        if (approved) {
            save.setExtensionUntil(save.round() + EXTENSION_GRACE_ROUNDS);
            save.setExtensionsGranted(save.extensionsGranted() + 1);
        } else {
            save.addHappiness(EXTENSION_DENIED_HAPPINESS);
        }
        saves.update(save);
        return new RentExtension(approved ? RentExtension.Status.APPROVED
                : RentExtension.Status.DENIED, save.extensionUntil());
    }

    /**
     * Switch to the other apartment type at today's listing, which becomes the new
     * locked-in rent. One month is bought; prior advances are forfeited; debt carries.
     */
    public ApartmentSwitch switchApartment(SaveState save) {
        ApartmentTier target = save.apartmentTier().other();
        int price = listing(target, save);
        ApartmentSwitch.Status status = null;
        if (!servicesOpen(save)) {
            status = ApartmentSwitch.Status.OFFICE_CLOSED;
        } else if (save.weekOver()) {
            status = ApartmentSwitch.Status.WEEK_OVER;
        } else if (save.cash() < price) {
            status = ApartmentSwitch.Status.INSUFFICIENT_CASH;
        } else if (save.timeMinutes() < costs.switchApartmentMinutes()) {
            status = ApartmentSwitch.Status.INSUFFICIENT_TIME;
        }
        if (status != null) {
            return new ApartmentSwitch(status, save.apartmentTier(), price,
                    save.leasePaidThrough(), save.timeMinutes(), save.cash());
        }
        save.setTimeMinutes(save.timeMinutes() - costs.switchApartmentMinutes());
        save.setCash(save.cash() - price);
        save.setApartmentTier(target);
        save.setRentMonthly(price);
        save.setLeasePaidThrough(nextMonthEndAfter(save.round()));
        save.setExtensionUntil(null);
        saves.update(save);
        return new ApartmentSwitch(ApartmentSwitch.Status.OK, target, price,
                save.leasePaidThrough(), save.timeMinutes(), save.cash());
    }
}
```
